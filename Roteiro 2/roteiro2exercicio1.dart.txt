import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      home: TelaInicial(),
    );
  }
}

class TelaInicial extends StatelessWidget {
  const TelaInicial({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("VOCÊ ESTÁ NA TELA INICIAL."),
      ),
      body: Center(
        child: ElevatedButton(
          child: const Text("IR PARA DETALHE."),
          onPressed: () {
            Navigator.push(
              context,
              MaterialPageRoute(
                builder: (context) => const TelaDetalhe(),
              ),
            );
          },
        ),
      ),
    );
  }
}

class TelaDetalhe extends StatelessWidget {
  const TelaDetalhe({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("VOCÊ ESTÁ NA TELA DETALHE"),
      ),
      body: Center(
        child: ElevatedButton(
          child: const Text("VOLTAR"),
          onPressed: () {
            Navigator.pop(context);
          },
        ),
      ),
    );
  }
}
